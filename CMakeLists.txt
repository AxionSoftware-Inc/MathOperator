cmake_minimum_required(VERSION 3.20)
project(opforge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OPFORGE_BUILD_TESTS "Build tests" ON)
add_library(opforge_core src/atlas/atlas.cpp src/atlas/validation.cpp src/atlas/loader.cpp src/atlas/geometry.cpp src/discovery/composition.cpp src/atlas/seed.cpp src/patterns/analyzer.cpp src/patterns/meta.cpp src/synthesis/candidate.cpp src/synthesis/registry.cpp src/synthesis/goal.cpp src/synthesis/schema.cpp src/research/evaluation.cpp src/research/residual.cpp src/research/oracles.cpp src/research/structure_analysis.cpp src/research/scientific.cpp src/research/deep.cpp src/research/campaign.cpp src/axiomatic/engine.cpp src/axiomatic/unknown.cpp src/semantic/closure.cpp src/analogy/engine.cpp src/numerics/executor.cpp src/numerics/adversarial.cpp src/numerics/truth.cpp src/numerics/curved.cpp src/benchmarks/rediscovery.cpp src/benchmarks/residual.cpp src/benchmarks/schema.cpp src/benchmarks/geometry.cpp)
target_include_directories(opforge_core PUBLIC include)
target_compile_options(opforge_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(opforge_demo src/main.cpp)
target_link_libraries(opforge_demo PRIVATE opforge_core)
add_executable(opforge src/cli/main.cpp)
target_link_libraries(opforge PRIVATE opforge_core)
if(OPFORGE_BUILD_TESTS)
  enable_testing()
  add_executable(opforge_tests tests/atlas_tests.cpp)
  target_link_libraries(opforge_tests PRIVATE opforge_core)
  add_test(NAME opforge_tests COMMAND opforge_tests)
  set_tests_properties(opforge_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
