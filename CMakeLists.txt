cmake_minimum_required(VERSION 3.20)
project(opforge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OPFORGE_BUILD_TESTS "Build tests" ON)

target_compile_options(opforge_core PRIVATE -Wall -Wextra -Wpedantic)
# The quotient path is intentionally exercised at million-scale in the
# deterministic stress benchmark; keep its streaming hot path optimized even
# when the repository's default build type is unoptimized.
set_source_files_properties(src/search/quotient.cpp PROPERTIES COMPILE_OPTIONS "-O2")

target_link_libraries(opforge_demo PRIVATE opforge_core)

target_link_libraries(opforge PRIVATE opforge_core)
